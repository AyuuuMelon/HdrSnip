cmake_minimum_required(VERSION 3.20)
project(HdrSnip LANGUAGES CXX RC)
option(HDRSNIP_BUILD_TESTS "Build diagnostic commands and tests" OFF)
add_executable(HdrSnip WIN32 main.cpp app.rc)
target_compile_features(HdrSnip PRIVATE cxx_std_17)
target_compile_definitions(HdrSnip PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(HdrSnip PRIVATE /utf-8 /W4 /EHsc)
set_property(TARGET HdrSnip PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
target_link_libraries(HdrSnip PRIVATE d3d11 dxgi windowscodecs ole32 oleaut32 shell32 shlwapi comctl32 user32 gdi32)
if(HDRSNIP_BUILD_TESTS)
    target_compile_definitions(HdrSnip PRIVATE HDRSNIP_TEST_BUILD)
    target_link_libraries(HdrSnip PRIVATE psapi dwmapi)
    enable_testing()
    add_test(NAME core COMMAND HdrSnip --self-test "${CMAKE_CURRENT_BINARY_DIR}/test-output")
endif()
target_link_options(HdrSnip PRIVATE "/MANIFEST:EMBED" "/MANIFESTINPUT:${CMAKE_CURRENT_SOURCE_DIR}/app.manifest" "/PDBALTPATH:HdrSnip.pdb")
